/* ---------- COMMON HELPERS ---------- */

function jsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function getDatabase() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (!ss) {
    throw new Error(
      "No spreadsheet is connected. Open this script from your Google Sheet."
    );
  }

  return ss;
}

function getSheet(sheetName) {
  const sheet = getDatabase().getSheetByName(sheetName);

  if (!sheet) {
    throw new Error("Sheet not found: " + sheetName);
  }

  return sheet;
}


/* ---------- CREATE DATABASE SHEETS ---------- */

function setupTarunnoSheets() {
  const ss = getDatabase();

  Object.entries(SHEET_HEADERS).forEach(function(entry) {
    const sheetName = entry[0];
    const headers = entry[1];

    let sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }

    if (sheet.getLastRow() === 0) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.setFrozenRows(1);
    } else {
      // Add newly introduced columns to existing sheets without changing existing data.
      const current = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];
      const missing = headers.filter(function(header) { return current.indexOf(header) === -1; });
      if (missing.length) sheet.getRange(1, sheet.getLastColumn() + 1, 1, missing.length).setValues([missing]);
    }
  });

  return "Tarunno database sheets are ready.";
}


/* ---------- READ SHEET DATA ---------- */

function rowsAsObjects(sheetName) {
  const sheet = getSheet(sheetName);
  const values = sheet.getDataRange().getDisplayValues();

  if (values.length < 2) {
    return [];
  }

  const headers = values[0];

  return values.slice(1)
    .filter(function(row) {
      return row.some(function(value) {
        return value !== "";
      });
    })
    .map(function(row) {
      const item = {};

      headers.forEach(function(header, index) {
        item[header] = row[index] || "";
      });

      return item;
    })
    .filter(function(item) {
      // Donor search performs its own approval + consent checks in the browser.
      if (sheetName === "BloodDonors") return true;
      return !item.status || String(item.status).toLowerCase() === "active";
    });
}

function escapeHtml_(value) {
  return String(value == null ? "" : value).replace(/[&<>"']/g, function(c) {
    return ({"&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;"})[c];
  });
}

/* ---------- GET API ---------- */

function doGet(e) {
  try {
    const params = e && e.parameter ? e.parameter : {};
    const action = params.action || "health";

    if (action === "health") {
      return jsonResponse({
        success: true,
        message: "Tarunno API is running"
      });
    }

    const publicSheets = [
      "SiteContent", "Projects", "Programs", "Members", "Gallery", "NewsEvents", "BloodDonors", "DonationMethods", "TeamMembers"
    ];
    const requestedSheet = action === "Donors" ? "BloodDonors" : action;

    if (publicSheets.indexOf(requestedSheet) === -1) {
      return jsonResponse({
        success: false,
        error: "UNKNOWN_ACTION",
        allowedActions: publicSheets
      });
    }

    return jsonResponse({
      success: true,
      data: rowsAsObjects(requestedSheet)
    });

  } catch (error) {
    return jsonResponse({
      success: false,
      error: String(error.message || error)
    });
  }
}


/* ---------- POST API: PUBLIC FORMS ---------- */

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({
        success: false,
        error: "MISSING_BODY"
      });
    }

    let body;

    try {
      body = JSON.parse(e.postData.contents);
    } catch (parseError) {
      return jsonResponse({
        success: false,
        error: "INVALID_JSON"
      });
    }

    const action = String(body.action || "");
    const data = body.data || {};

    // Admin-managed donation methods and About Us team directory. Configure Script Property TARUNNO_ADMIN_KEY before use.
    const adminActions = ["saveDonationMethod", "deleteDonationMethod", "saveTeamMember", "deleteTeamMember"];
    if (adminActions.indexOf(action) !== -1) {
      const expectedKey = PropertiesService.getScriptProperties().getProperty("TARUNNO_ADMIN_KEY");
      if (!expectedKey || String(body.adminKey || "") !== expectedKey) return jsonResponse({success:false,error:"ADMIN_AUTH_REQUIRED"});
      const teamAction = action.indexOf("TeamMember") !== -1;
      const targetSheet = getSheet(teamAction ? "TeamMembers" : "DonationMethods");
      const headers = targetSheet.getRange(1,1,1,targetSheet.getLastColumn()).getDisplayValues()[0];
      const rows = rowsAsObjects(teamAction ? "TeamMembers" : "DonationMethods");
      const id = String(data.id || "").trim();
      if (action.indexOf("delete") === 0) {
        const rowIndex = rows.findIndex(function(r){return String(r.id) === id;});
        if (rowIndex < 0) return jsonResponse({success:false,error:"RECORD_NOT_FOUND"});
        targetSheet.deleteRow(rowIndex + 2);
        return jsonResponse({success:true,deleted:id});
      }
      const required = teamAction ? ["category","name","designation"] : ["method","account"];
      for (let i=0;i<required.length;i++) if (!String(data[required[i]] || "").trim()) return jsonResponse({success:false,error:"MISSING_FIELD",field:required[i]});
      const record = {id:id || Utilities.getUuid(), updatedAt:new Date().toISOString(), status:String(data.status || "active")};
      if (teamAction) {
        ["category","name","designation","photo"].forEach(function(k){record[k]=String(data[k] || "").trim();});
        if (data.photoData && String(data.photoData).indexOf("data:image/") === 0) {
          const parts = String(data.photoData).match(/^data:(image\/(?:png|jpeg|webp));base64,([\s\S]+)$/);
          if (!parts) return jsonResponse({success:false,error:"INVALID_IMAGE"});
          const bytes = Utilities.base64Decode(parts[2]);
          if (bytes.length > 1500000) return jsonResponse({success:false,error:"IMAGE_TOO_LARGE",message:"Please use an image under 1.5 MB."});
          const blob = Utilities.newBlob(bytes, parts[1], "tarunno-team-" + record.id + (parts[1] === "image/png" ? ".png" : ".jpg"));
          const folderName = "Tarunno Team Photos";
          const folders = DriveApp.getFoldersByName(folderName);
          const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
          const file = folder.createFile(blob);
          file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
          record.photo = "https://drive.google.com/uc?export=view&id=" + file.getId();
        }
      } else {
        ["method","account","accountName"].forEach(function(k){record[k]=String(data[k] || "").trim();});
      }
      const existingIndex = id ? rows.findIndex(function(r){return String(r.id) === id;}) : -1;
      const values = headers.map(function(h){return record[h] === undefined ? (existingIndex >= 0 ? rows[existingIndex][h] || "" : "") : record[h];});
      if (existingIndex >= 0) targetSheet.getRange(existingIndex + 2,1,1,headers.length).setValues([values]); else targetSheet.appendRow(values);
      return jsonResponse({success:true,data:record});
    }

    const forms = {
      registerDonor: {
        sheet: "BloodDonors",
        required: ["name", "bloodGroup", "phone", "district", "upazila", "gender", "dateOfBirth", "consent"],
        fields: ["name", "bloodGroup", "phone", "district", "upazila", "lastDonation", "consent", "email", "gender", "dateOfBirth", "address"]
      },

      submitDonation: {
        sheet: "Donations",
        required: [
          "donorName", "method", "amount", "transactionId"
        ],
        fields: [
          "donorName", "phone", "method",
          "amount", "transactionId"
        ]
      },

      registerMember: {
        sheet: "Members",
        required: ["name", "phone", "gender", "dateOfBirth", "address"],
        fields: ["name", "phone", "email", "skills", "gender", "dateOfBirth", "address", "photoData", "language"]
      },
      registerVolunteer: {
        sheet: "Volunteers",
        required: ["name", "phone"],
        fields: ["name", "phone", "email", "skills", "gender", "dateOfBirth", "address", "applicationType"]
      }
    };

    const form = forms[action];

    if (!form) {
      return jsonResponse({
        success: false,
        error: "UNKNOWN_ACTION",
        allowedActions: Object.keys(forms)
      });
    }

    // Validate required fields.
    for (let i = 0; i < form.required.length; i++) {
      const field = form.required[i];

      if (
        data[field] === undefined ||
        data[field] === null ||
        String(data[field]).trim() === ""
      ) {
        return jsonResponse({
          success: false,
          error: "MISSING_FIELD",
          field: field
        });
      }
    }

    // Donor registration requires explicit consent.
    if (
      action === "registerDonor" &&
      String(data.consent).toLowerCase() !== "true"
    ) {
      return jsonResponse({
        success: false,
        error: "CONSENT_REQUIRED"
      });
    }

    // Validate donation details.
    if (action === "submitDonation") {
      const amount = Number(data.amount);
      const allowedMethods = ["Bkash", "Nagad", "Rocket"];

      if (!Number.isFinite(amount) || amount <= 0) {
        return jsonResponse({
          success: false,
          error: "INVALID_AMOUNT"
        });
      }

      if (allowedMethods.indexOf(String(data.method)) === -1) {
        return jsonResponse({
          success: false,
          error: "INVALID_PAYMENT_METHOD"
        });
      }
    }

    const sheet = getSheet(form.sheet);

    const record = {
      id: Utilities.getUuid(),
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      status: "pending"
    };

    form.fields.forEach(function(field) {
      record[field] = String(data[field] ?? "").trim();
    });
    if (action === "registerMember") {
      record.role = "Applicant";
      record.category = "Membership application";
      record.photo = record.photoData || "";
      delete record.photoData;
      record.status = "pending";
      record.language = String(data.language || "bn");
    }
    // Normalize Bangladeshi mobile numbers that were entered without the leading zero.
    if (record.phone) {
      let phone = record.phone.replace(/[\s()-]/g, "");
      if (/^1[3-9]\d{8}$/.test(phone)) phone = "0" + phone;
      if (/^8801[3-9]\d{8}$/.test(phone)) phone = "0" + phone.slice(3);
      record.phone = phone;
    }

    // Use the actual sheet header order so newly appended columns remain aligned.
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getDisplayValues()[0];
    sheet.appendRow(headers.map(function(header) { return record[header] ?? ""; }));

    // Branded HTML confirmation email. Email is sent by the account that owns/deploys this script.
    const recipient = String(data.email || "").trim();
    if (recipient && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(recipient)) {
      const name = String(data.name || "বন্ধু").trim() || "বন্ধু";
      const isBn = String(data.language || "bn").toLowerCase().startsWith("bn");
      let subject = "", heading = "", intro = "", detail = "";

        intro = isBn ? "আপনার সদস্যপদ আবেদন আমরা পেয়েছি। আমাদের টিম তথ্য যাচাই করে সিদ্ধান্ত জানাবে। অনুমোদনের পর আলাদা Welcome Email-এ আপনার Membership Number পাঠানো হবে।" : "We have received your membership application.
